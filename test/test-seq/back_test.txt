# Backward scrolling 4-bit weighted sequencer
# Mirrors the forward file but walks memory upward → downward (net -1 row per frame)
# Assumptions:
# - Each bitplane spans 1080 rows and lives in inums separated by StepInum
# - We start from the last 1080-row block (StartImageNo = 2 for 3 blocks total)
# - MemStartRow is zero-indexed; for 1080 rows valid indices are 0–1079

AssignVar  RowsPrGrp       72 1    # 1+2+4+8=15 groups over 1080 rows => 1080/15=72
AssignVar  NoOfDMDrows   1080 1    # DMD size in rows
AssignVar  NoOfInums        3 1    # 3 blocks (inum 0,1,2) of 1080 rows each
AssignVar  MemStartRow   1008 1    # Start at bottom block: rows 1008–1079
AssignVar  StartImageNo     2 1    # Start from last inum block
AssignVar  StepInum      1699 1    # Inum separation between bit-planes.

Label Scroll 1

    ######################## 0 ###################################
    AssignVar DmdStartRow             0               1
    AssignVar inum                    0               1
    Add       inum         StartImageNo               1    
    LoadRow   DmdStartRow  RowsPrGrp inum MemStartRow 11

    ######################## 1 ###################################
    Add       inum        StepInum                    1
    Add       MemStartRow -RowsPrGrp                  1
    Add       DmdStartRow RowsPrGrp                   1
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       MemStartRow -RowsPrGrp                  1
    Add       DmdStartRow RowsPrGrp                   1
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  11

    ######################## 2 ###################################
    Add       inum        StepInum                    1  
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1 
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1         
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1 
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1 
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  11

    ######################## 3 ###################################
    Add       inum        StepInum                    1  
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1 
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1 
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1     
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1 
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1 
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1     
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1     
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
    Add       DmdStartRow RowsPrGrp                   1
    Add       MemStartRow -RowsPrGrp                  1     
    LoadRow   DmdStartRow RowsPrGrp inum MemStartRow  12
       
  Trig 0 4 0

  ResetGlobal               40
  LightPulseWord  15 1000    # Adjust pulse length as needed
  Add MemStartRow          -1    1   # Net -1 row per frame

JumpIf MemStartRow >= 0 Scroll 1
  Add       StartImageNo -1          1  # Move to previous inum block
  Add       MemStartRow  1080        1  # Wrap MemStartRow back into valid range (e.g., -1 -> 1079)

JumpIf StartImageNo >= 0 Scroll 1

Label Stop 1
Jump Stop 1
